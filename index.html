// only forward transitions are allowed — everything else is rejected
private static readonly Dictionary<OrderItemStatus, OrderItemStatus[]> AllowedTransitions = new()
{
    [Processing] = new[] { Shipped, Cancelled },
    [Shipped]    = new[] { Delivered, Cancelled },
    [Delivered]  = Array.Empty<OrderItemStatus>(),
    [Cancelled]  = Array.Empty<OrderItemStatus>(),
};

public async Task ChangeOrderItemStatus(int sellerBookId, int orderId,
    OrderItemStatus newStatus, int sellerId)
{
    var item = await _repo.GetOrderItemByIds(sellerBookId, orderId);

    // resource-level check — role alone can't stop cross-seller access
    if (item.SellerBook.SellerId != sellerId)
        throw new UnauthorizedAccessException("Not your order item.");

    if (!AllowedTransitions.TryGetValue(item.Status, out var next)
        || !next.Contains(newStatus))
        throw new InvalidOperationException(
            $"Cannot move from '{item.Status}' to '{newStatus}'.");

    item.Status = newStatus;
    if (newStatus == Shipped)
        item.TrackingCode = await GenerateUniqueTrackingCodeAsync();

    await _workRepo.SaveChangesAsync();
}
